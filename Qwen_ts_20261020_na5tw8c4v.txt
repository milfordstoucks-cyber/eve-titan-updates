export function getItemName(typeId: number, lang: Language): string {
  return ITEM_NAMES[typeId]?.[lang] ?? `Item ${typeId}`;
}

export function getMineralName(typeId: number, lang: Language): string {
  return getItemName(typeId, lang);
}

export function getComponentName(typeId: number, lang: Language): string {
  return getItemName(typeId, lang);
}

export function getSkillName(typeId: number, lang: Language): string {
  return getItemName(typeId, lang);
}

export function getSkillDescription(typeId: number, lang: Language): string {
  return SKILL_DESCRIPTIONS[lang][typeId] ?? '';
}

const MINERAL_IDS = [34, 35, 36, 37, 38, 39, 40, 11399, 2872, 57457];

const buildNameMap = (ids: number[], lang: Language): Record<number, string> =>
  ids.reduce<Record<number, string>>((acc, id) => {
    acc[id] = getItemName(id, lang);
    return acc;
  }, {});
