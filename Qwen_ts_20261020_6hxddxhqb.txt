export const COMPONENT_MAP = new Map(
  COMPONENT_DEFS.map((component) => [component.typeId, component])
);

export function getComponentDef(typeId: number): ComponentDef | undefined {
  return COMPONENT_MAP.get(typeId);
}

export function buildMaterials(materials: Record<number, number>): Material[] {
  return Object.entries(materials).map(([typeIdString, quantity]) => {
    const typeId = Number(typeIdString);
    const def = MATERIAL_DEFS[typeId];

    return {
      typeId,
      name: getItemName(typeId, 'en'),
      nameRu: getItemName(typeId, 'ru'),
      quantity,
      icon: EVE_ICON(typeId),
      category: def?.category ?? 'mineral',
      schematicId: def?.schematicId,
      formulaTypeId: def?.formulaTypeId,
    };
  });
}

export function toComponentMaterial(
  def: ComponentDef,
  quantity: number
): ComponentMaterial {
  return {
    typeId: def.typeId,
    blueprintTypeId: def.blueprintTypeId,
    name: getItemName(def.typeId, 'en'),
    quantity,
    icon: EVE_ICON(def.typeId),
    baseTime: def.productionTime,
    materials: def.materials,
    category: def.category,
  };
}
